const API_BASE = "/api/tg-bot/admin"
function getAdminKey(): string { if (typeof window === "undefined") return ""; return localStorage.getItem("tg_admin_key") || "" }
async function api(action: string): Promise<Response> { return fetch(`${API_BASE}?action=${action}`, { headers: { "x-admin-key": getAdminKey() } }) }
async function apiPost(body: Record<string, unknown>): Promise<Response> { return fetch(API_BASE, { method: "POST", headers: { "Content-Type": "application/json", "x-admin-key": getAdminKey() }, body: JSON.stringify(body) }) }
